> 从 demo 到生产的最后一公里，也是最难的一公里。90% 的 Agent 项目死在这一步。

## 一、为什么部署这么难？

### 1.1 Demo 和生产的差距

| 维度 | Demo | 生产 |
|------|------|------|
| 并发 | 1 个用户 | 数百/数千并发 |
| 可靠性 | 偶尔出错没关系 | 99.9% 可用性 |
| 延迟 | 等几秒无所谓 | 用户等 3 秒就走了 |
| 成本 | 不在乎 | 每月 Token 费用可能上万 |
| 安全 | 本地跑 | 面向公网，防攻击 |
| 状态 | 内存存储 | 持久化，故障恢复 |

### 1.2 现实类比

Demo 就像在自家厨房做饭——食材随便放，火候随意调，做给自己吃。

生产就像开一家餐厅——要有标准化的菜谱（Prompt）、稳定的供应链（模型 API）、高效的厨房动线（架构）、食品安全检查（安全防护）、翻台率（性能）。

```python
from fastapi import FastAPI, HTTPException
from fastapi.responses import StreamingResponse
from pydantic import BaseModel, Field
from contextlib import asynccontextmanager
import uvicorn

# 全局 Agent 实例
agent_app = None

@asynccontextmanager
async def lifespan(app: FastAPI):
    """应用生命周期管理"""
    global agent_app
    # 启动时初始化 Agent
    agent_app = build_agent()
    print("Agent 初始化完成")
    yield
    # 关闭时清理资源
    print("Agent 资源已释放")

app = FastAPI(title="Enterprise Agent API", lifespan=lifespan)

class ChatRequest(BaseModel):
    message: str = Field(description="用户消息")
    thread_id: str = Field(description="会话 ID")
    user_id: str = Field(description="用户 ID")

class ChatResponse(BaseModel):
    response: str = Field(description="Agent 回复")
    thread_id: str
    tokens_used: int = 0
    tools_called: list[str] = []

@app.post("/chat", response_model=ChatResponse)
async def chat(request: ChatRequest):
    """与 Agent 对话"""
    try:
        config = {"configurable": {"thread_id": request.thread_id}}
        result = await agent_app.ainvoke(
            {"messages": [HumanMessage(content=request.message)]},
            config=config,
        )
        last_msg = result["messages"][-1]
        return ChatResponse(
            response=last_msg.content,
            thread_id=request.thread_id,
            tokens_used=getattr(last_msg, "usage_metadata", {}).get("total_tokens", 0),
        )
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))
```

### 2.2 流式输出 API

```python
@app.post("/chat/stream")
async def chat_stream(request: ChatRequest):
    """流式对话 —— 实时返回 Agent 的思考和回答过程"""

    async def event_generator():
        config = {"configurable": {"thread_id": request.thread_id}}
        async for event in agent_app.astream_events(
            {"messages": [HumanMessage(content=request.message)]},
            config=config,
            version="v2",
        ):
            kind = event["event"]
            if kind == "on_chat_model_stream":
                token = event["data"]["chunk"].content
                if token:
                    yield f"data: {json.dumps({'type': 'token', 'content': token}, ensure_ascii=False)}\n\n"
            elif kind == "on_tool_start":
                yield f"data: {json.dumps({'type': 'tool_start', 'name': event['name']}, ensure_ascii=False)}\n\n"
            elif kind == "on_tool_end":
                yield f"data: {json.dumps({'type': 'tool_end', 'name': event['name']}, ensure_ascii=False)}\n\n"
        yield "data: [DONE]\n\n"

    return StreamingResponse(
        event_generator(),
        media_type="text/event-stream",
        headers={"Cache-Control": "no-cache", "Connection": "keep-alive"},
    )
```

```python
import asyncio

async def parallel_tool_execution(tools_to_call: list[dict]) -> list[str]:
    """并行执行多个工具调用"""
    tasks = []
    for tool_call in tools_to_call:
        task = asyncio.create_task(
            execute_tool_async(tool_call["name"], tool_call["args"])
        )
        tasks.append(task)

    results = await asyncio.gather(*tasks, return_exceptions=True)

    processed = []
    for result in results:
        if isinstance(result, Exception):
            processed.append(f"工具执行失败: {result}")
        else:
            processed.append(result)
    return processed
```

### 3.2 缓存策略

```python
from functools import lru_cache
import hashlib
import json

class AgentCache:
    """Agent 响应缓存 —— 对相同问题避免重复调用 LLM"""

    def __init__(self, max_size: int = 1000, ttl_seconds: int = 3600):
        self.cache: dict[str, tuple[str, float]] = {}
        self.max_size = max_size
        self.ttl = ttl_seconds

    def _make_key(self, messages: list) -> str:
        content = json.dumps([m.content for m in messages], ensure_ascii=False)
        return hashlib.md5(content.encode()).hexdigest()

    def get(self, messages: list) -> str | None:
        key = self._make_key(messages)
        if key in self.cache:
            value, timestamp = self.cache[key]
            if time.time() - timestamp < self.ttl:
                return value
            del self.cache[key]
        return None

    def set(self, messages: list, response: str):
        if len(self.cache) >= self.max_size:
            # 淘汰最旧的
            oldest_key = min(self.cache, key=lambda k: self.cache[k][1])
            del self.cache[oldest_key]
        key = self._make_key(messages)
        self.cache[key] = (response, time.time())

# 在 Agent 节点中使用缓存
cache = AgentCache()

def cached_agent(state: MessagesState) -> dict:
    cached = cache.get(state["messages"])
    if cached:
        return {"messages": [AIMessage(content=cached)]}

    response = llm.invoke(state["messages"])
    cache.set(state["messages"], response.content)
    return {"messages": [response]}
```

### 3.4 速率限制

```python
import asyncio
from collections import deque
from time import time

class RateLimiter:
    """令牌桶速率限制器"""

    def __init__(self, max_requests: int, time_window: int = 60):
        self.max_requests = max_requests
        self.time_window = time_window
        self.requests: deque[float] = deque()
        self.lock = asyncio.Lock()

    async def acquire(self):
        async with self.lock:
            now = time()
            # 清理过期的请求记录
            while self.requests and self.requests[0] < now - self.time_window:
                self.requests.popleft()

            if len(self.requests) >= self.max_requests:
                wait_time = self.requests[0] + self.time_window - now
                raise HTTPException(
                    status_code=429,
                    detail=f"请求过于频繁，请 {wait_time:.1f} 秒后重试"
                )

            self.requests.append(now)

rate_limiter = RateLimiter(max_requests=100, time_window=60)

@app.post("/chat")
async def chat(request: ChatRequest):
    await rate_limiter.acquire()
    # ... 处理请求
```

### 5.2 输入输出过滤

```python
class ContentFilter:
    """内容安全过滤器"""

    SENSITIVE_PATTERNS = [
        r'\b\d{17}[\dXx]\b',           # 身份证号
        r'\b1[3-9]\d{9}\b',            # 手机号
        r'\b\d{16,19}\b',              # 银行卡号
        r'[\w.+-]+@[\w-]+\.[\w.-]+',   # 邮箱
    ]

    @classmethod
    def filter_output(cls, text: str) -> str:
        """过滤输出中的敏感信息"""
        import re
        for pattern in cls.SENSITIVE_PATTERNS:
            text = re.sub(pattern, '***', text)
        return text
```
