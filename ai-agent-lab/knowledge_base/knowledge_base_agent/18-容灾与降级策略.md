# 容灾与降级策略

> 适用场景：依赖外部 LLM API 的生产 Agent 系统，需要应对 API 故障、超时、限流等异常
> 最后更新：2026 年 4 月

---

## 一、为什么需要容灾

生产环境中，LLM API 不是 100% 可靠的：

- **API 宕机**：OpenAI、Anthropic 等服务偶尔会出现故障
- **限流（Rate Limiting）**：请求过多被 API 拒绝
- **超时**：网络波动或服务端负载高导致响应超时
- **配额耗尽**：月度 token 配额用完
- **模型下线**：某个模型版本被废弃

**现实例子**：2024 年某次 OpenAI 大规模故障持续数小时，所有依赖 GPT-4 的 Agent 系统全部瘫痪。有容灾方案的企业在几秒内自动切换到备用模型，用户几乎无感知。

## 二、模型降级链

核心思路：主模型不可用时，自动降级到备用模型。

### 2.1 基本降级链

```python
from langchain_openai import ChatOpenAI
from langchain_anthropic import ChatAnthropic

class ModelFallbackChain:
    """模型降级链：按优先级尝试多个模型"""

    def __init__(self):
        self.models = [
            {
                "name": "gpt-4o",
                "instance": ChatOpenAI(model="gpt-4o", request_timeout=30),
                "priority": 1,
            },
            {
                "name": "claude-3.5-sonnet",
                "instance": ChatAnthropic(model="claude-3-5-sonnet-20241022", timeout=30),
                "priority": 2,
            },
            {
                "name": "gpt-4o-mini",
                "instance": ChatOpenAI(model="gpt-4o-mini", request_timeout=30),
                "priority": 3,
            },
        ]

    async def ainvoke(self, messages, **kwargs):
        """依次尝试每个模型，直到成功"""
        last_error = None

        for model_config in self.models:
            try:
                result = await model_config["instance"].ainvoke(messages, **kwargs)
                if model_config["priority"] > 1:
                    logger.warning(
                        f"主模型不可用，已降级到 {model_config['name']}"
                    )
                return result
            except Exception as e:
                last_error = e
                logger.error(
                    f"模型 {model_config['name']} 调用失败: {e}"
                )
                continue

        # 所有模型都失败
        raise RuntimeError(f"所有模型均不可用，最后错误: {last_error}")
```

### 2.2 使用 LangChain 内置的 with_fallbacks

```python
from langchain_openai import ChatOpenAI
from langchain_anthropic import ChatAnthropic

primary = ChatOpenAI(model="gpt-4o", request_timeout=30)
fallback_1 = ChatAnthropic(model="claude-3-5-sonnet-20241022", timeout=30)
fallback_2 = ChatOpenAI(model="gpt-4o-mini", request_timeout=30)

# 一行代码实现降级链
model_with_fallback = primary.with_fallbacks([fallback_1, fallback_2])

# 使用方式和普通模型完全一样
result = await model_with_fallback.ainvoke(messages)
```

## 三、熔断机制

当某个模型连续失败时，暂时停止向它发送请求，避免浪费时间和资源。

### 3.1 熔断器实现

```python
import time
from enum import Enum

class CircuitState(Enum):
    CLOSED = "closed"      # 正常状态，请求正常通过
    OPEN = "open"          # 熔断状态，请求直接跳过
    HALF_OPEN = "half_open"  # 半开状态，允许少量请求试探

class CircuitBreaker:
    """熔断器"""

    def __init__(
        self,
        failure_threshold: int = 5,     # 连续失败 N 次触发熔断
        recovery_timeout: float = 60.0, # 熔断后等待 N 秒再试探
    ):
        self.failure_threshold = failure_threshold
        self.recovery_timeout = recovery_timeout
        self.failure_count = 0
        self.state = CircuitState.CLOSED
        self.last_failure_time = 0

    def can_execute(self) -> bool:
        """是否允许执行请求"""
        if self.state == CircuitState.CLOSED:
            return True

        if self.state == CircuitState.OPEN:
            # 检查是否到了试探时间
            if time.time() - self.last_failure_time > self.recovery_timeout:
                self.state = CircuitState.HALF_OPEN
                return True
            return False

        # HALF_OPEN：允许一个请求试探
        return True

    def record_success(self):
        """记录成功"""
        self.failure_count = 0
        self.state = CircuitState.CLOSED

    def record_failure(self):
        """记录失败"""
        self.failure_count += 1
        self.last_failure_time = time.time()

        if self.failure_count >= self.failure_threshold:
            self.state = CircuitState.OPEN
            logger.warning(
                f"熔断器触发：连续失败 {self.failure_count} 次，"
                f"将在 {self.recovery_timeout} 秒后重试"
            )
```

### 3.2 熔断器 + 降级链

```python
class ResilientModelRouter:
    """带熔断器的模型路由"""

    def __init__(self):
        self.models = {
            "gpt-4o": ChatOpenAI(model="gpt-4o"),
            "claude-3.5-sonnet": ChatAnthropic(model="claude-3-5-sonnet-20241022"),
            "gpt-4o-mini": ChatOpenAI(model="gpt-4o-mini"),
        }
        self.breakers = {
            name: CircuitBreaker() for name in self.models
        }
        self.priority = ["gpt-4o", "claude-3.5-sonnet", "gpt-4o-mini"]

    async def ainvoke(self, messages, **kwargs):
        for model_name in self.priority:
            breaker = self.breakers[model_name]

            if not breaker.can_execute():
                logger.info(f"跳过 {model_name}（熔断中）")
                continue

            try:
                result = await self.models[model_name].ainvoke(messages, **kwargs)
                breaker.record_success()
                return result
            except Exception as e:
                breaker.record_failure()
                logger.error(f"{model_name} 失败: {e}")
                continue

        raise RuntimeError("所有模型均不可用或已熔断")
```

## 四、重试策略

### 4.1 指数退避重试

```python
import asyncio
import random

async def retry_with_backoff(
    func,
    max_retries: int = 3,
    base_delay: float = 1.0,
    max_delay: float = 30.0,
):
    """指数退避重试"""
    for attempt in range(max_retries + 1):
        try:
            return await func()
        except Exception as e:
            if attempt == max_retries:
                raise

            # 指数退避 + 随机抖动
            delay = min(base_delay * (2 ** attempt), max_delay)
            jitter = random.uniform(0, delay * 0.1)
            actual_delay = delay + jitter

            logger.warning(
                f"第 {attempt + 1} 次重试，等待 {actual_delay:.1f} 秒: {e}"
            )
            await asyncio.sleep(actual_delay)
```

### 4.2 LangChain 内置重试

## 五、限流保护

### 5.1 API 层限流

```python
from fastapi import Request
from collections import defaultdict
import time

class RateLimiter:
    """简单的滑动窗口限流器"""

    def __init__(self, max_requests: int = 10, window_seconds: int = 60):
        self.max_requests = max_requests
        self.window_seconds = window_seconds
        self.requests = defaultdict(list)  # user_id -> [timestamps]

    def is_allowed(self, user_id: str) -> bool:
        now = time.time()
        window_start = now - self.window_seconds

        # 清理过期记录
        self.requests[user_id] = [
            t for t in self.requests[user_id] if t > window_start
        ]

        if len(self.requests[user_id]) >= self.max_requests:
            return False

        self.requests[user_id].append(now)
        return True

rate_limiter = RateLimiter(max_requests=20, window_seconds=60)

@app.post("/chat")
async def chat(request: ChatRequest):
    if not rate_limiter.is_allowed(request.user_id):
        raise HTTPException(
            status_code=429,
            detail="请求过于频繁，请稍后再试"
        )
    ...
```

## 六、优雅降级

当系统部分功能不可用时，提供有限但可用的服务，而不是完全瘫痪。

```python
async def chat_with_graceful_degradation(message: str, config: dict):
    """优雅降级的聊天接口"""

    # 第一级：完整功能（RAG + 工具 + GPT-4）
    try:
        return await full_agent.ainvoke(
            {"messages": [("user", message)]}, config
        )
    except Exception as e:
        logger.error(f"完整 Agent 失败: {e}")

    # 第二级：简化功能（仅 RAG，无工具，GPT-4o-mini）
    try:
        context = rag_engine.query(message)
        return await simple_llm.ainvoke(
            f"基于以下上下文回答问题：\n{context}\n\n问题：{message}"
        )
    except Exception as e:
        logger.error(f"简化 Agent 失败: {e}")

    # 第三级：静态回复
    return {
        "messages": [AIMessage(content=(
            "抱歉，系统当前繁忙，无法处理您的请求。"
            "请稍后重试，或联系人工客服：400-xxx-xxxx"
        ))]
    }
```

## 七、健康检查与告警

```python
@app.get("/health")
async def health_check():
    """系统健康检查"""
    checks = {}

    # 检查 LLM API
    try:
        await model.ainvoke([("user", "ping")])
        checks["llm_api"] = "healthy"
    except Exception:
        checks["llm_api"] = "unhealthy"

    # 检查向量数据库
    try:
        vectorstore.similarity_search("test", k=1)
        checks["vector_db"] = "healthy"
    except Exception:
        checks["vector_db"] = "unhealthy"

    # 检查 Checkpoint 数据库
    try:
        # 简单的连接测试
        checks["checkpoint_db"] = "healthy"
    except Exception:
        checks["checkpoint_db"] = "unhealthy"

    overall = "healthy" if all(v == "healthy" for v in checks.values()) else "degraded"

    return {"status": overall, "checks": checks}
```
